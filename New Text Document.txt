# Q1: Gauss-Jordan Elimination
import numpy as np

def find_pivot(A, col, start_row):

    if np.allclose(A[start_row:, col], 0):
        return None

    pivot_row = start_row + np.argmax(
        np.abs(A[start_row:, col])
    )

    return pivot_row

def pivot_matrix(A, col, start_row):

    pivot_row = find_pivot(A, col, start_row)

    if pivot_row is None:
        return None

    if pivot_row != start_row:
        A[[start_row, pivot_row]] = A[[pivot_row, start_row]]

    return A

def normalize_pivot(A, row, col):
    pivot = A[row, col]

    A[row, :] = A[row, :] / pivot

    return A

def eliminate_column(A, row, col):
    n = A.shape[0]

    for i in range(n):

        if i != row:

            multiplier = A[i, col]

            A[i, :] = A[i, :] - multiplier * A[row, :]

    return A

def gauss_jordan(A, verbose=True):
    A = A.astype(float)

    n = A.shape[0]

    I = np.eye(n)

    Aug = np.hstack((A, I))

    if verbose:
        print("Initial Augmented Matrix [A | I]:")
        print(Aug)
        print()

    for k in range(n):

        if verbose:
            print(f"Step {k+1}: Working on column {k}")

        pivot_row = find_pivot(Aug, k, k)

        if pivot_row is None:

            if verbose:
                print("No pivot found in this column.")
                print()

            continue

        if verbose:
            print(
                f"Pivot located at row {pivot_row} "
                f"with value {Aug[pivot_row, k]}"
            )

        if pivot_row != k:

            if verbose:
                print(
                    f"Swapping Row {k} with Row {pivot_row}"
                )

            Aug[[k, pivot_row]] = Aug[[pivot_row, k]]

            if verbose:
                print("Matrix after pivoting:")
                print(Aug)
                print()

        pivot = Aug[k, k]

        if verbose:
            print(f"Normalizing Row {k}")
            print(f"Pivot = {pivot}")

        Aug = normalize_pivot(Aug, k, k)

        if verbose:
            print("Matrix after normalization:")
            print(Aug)
            print()

        for i in range(n):

            if i != k:

                multiplier = Aug[i, k]

                if verbose:
                    print(
                        f"Eliminating entry at position "
                        f"({i},{k})"
                    )
                    print(f"Multiplier = {multiplier}")

                Aug[i, :] = (
                    Aug[i, :]
                    - multiplier * Aug[k, :]
                )

                if verbose:
                    print("Updated Matrix:")
                    print(Aug)
                    print()

    if verbose:
        print("Gauss-Jordan Elimination Completed.")
        print()
        print("Final Matrix [I | A^-1]:")
        print(Aug)
        print()

    inverse = Aug[:, n:]

    return inverse

A = np.array([
    [0.0, 3.0, 2.0],
    [4.0, 1.0, 1.0],
    [2.0, 5.0, 3.0]
])

A_inverse = gauss_jordan(A, verbose=True)

print("Inverse of A:")
print(A_inverse)

verification = np.matmul(A, A_inverse)

print()
print("Verification: A × A^-1")
print(verification)

print()
print("Is A × A^-1 approximately equal to Identity?")
print(np.allclose(verification, np.eye(len(A))))

# Q2: Gaussian Elimination
import numpy as np

A = np.array([
    [1.0,  2.0,  3.0, 14.0],
    [2.0, -1.0,  1.0,  3.0],
    [3.0,  1.0, -2.0, 4.0]
])

n = len(A)

print("Initial Augmented Matrix:")
print(A)



for i in range(n):

    if abs(A[i][i]) < 1e-12:

        for j in range(i + 1, n):
            if abs(A[j][i]) > 1e-12:
                A[[i, j]] = A[[j, i]]

                print(f"\nR{i+1} <-> R{j+1}")
                print(A)
                break

    for j in range(i + 1, n):

        factor = A[j][i] / A[i][i]

        A[j] = A[j] - factor * A[i]

        print(f"\nR{j+1} -> R{j+1} - ({factor})R{i+1}")
        print(A)

print("\nUpper Triangular Matrix:")
print(A)


x = np.zeros(n)

for i in range(n - 1, -1, -1):

    sum_value = 0

    for j in range(i + 1, n):
        sum_value += A[i][j] * x[j]

    x[i] = (A[i][n] - sum_value) / A[i][i]


print("\nSolution:")
print("x =", x[0])
print("y =", x[1])
print("z =", x[2])


coefficients = np.array([
    [1.0, 2.0, 3.0],
    [2.0, -1.0, 1.0],
    [3.0, 1.0, -2.0]
])

constants = np.array([14.0, 3.0, 4.0])

numpy_solution = np.linalg.solve(coefficients, constants)

print("\nVerification using NumPy:")
print(numpy_solution)

print("\nAre both solutions approximately equal?")
print(np.allclose(x, numpy_solution))



#3. Root Finding using Bisection Method

def f(x):
    return x**3 - 2*x - 5

a = 2
b = 3

tol = 1e-3

print("Bisection Method")

if f(a) * f(b) >= 0:
    print("Bisection method cannot be applied.")
else:

    iteration = 0

    while abs(b - a) > tol:

        iteration += 1

        c = (a + b) / 2

        print(
            f"Iteration {iteration}: "
            f"a = {a:.6f}, "
            f"b = {b:.6f}, "
            f"c = {c:.6f}, "
            f"f(c) = {f(c):.6f}"
        )

        if f(c) == 0:
            break

        if f(a) * f(c) < 0:
            b = c
        else:
            a = c

    root = (a + b) / 2

    print("Approximate Root =", root)
    print("Final Error =", abs(b - a))


# Q4: Newton-Raphson Method

def f(x):
    return x**3 - 3*x**2 - 8


def df(x):
    return 3*x**2 - 6*x


x = 1.5
tol = 1e-4

max_iterations = 100

print("Newton-Raphson Iterations")

for i in range(max_iterations):

    fx = f(x)
    dfx = df(x)

    if dfx == 0:
        print("Derivative became zero.")
        break

    x_new = x - fx / dfx

    print(
        f"Iteration {i+1}: "
        f"x = {x:.6f}, "
        f"f(x) = {fx:.6f}, "
        f"x_new = {x_new:.6f}"
    )


    if abs(x_new - x) < tol:
        x = x_new
        break

    x = x_new

print("Approximate Root =", x)
print("Final Error =", abs(x_new - x))
